86 void runBacktest(const string& data_path) {
87 vector<std::shared_ptr<Order>> current_orders;
88 vector<TimeType*> timetype_vector;
89 ifstream file(data_path);
90

92 cerr << "Error opening the file." << endl;

96 int n = 2;
97 string line;
98 getline(file, line); // Skip first line
99 while (getline(file, line)) {
100 vector<string> tokens;
101 boost::split(tokens, line, boost::is_any_of(","));
102
103 // Finding Exchange
104 std::shared_ptr<Exchange> exchange_ptr = user.findExchange(tokens[4]);
105
106 if (exchange_ptr == nullptr) {
107 throw runtime_error("Exchange " + tokens[4] + " is not found");
108 }
109
110 // Finding Security
111 std::shared_ptr<Security> security_ptr = exchange_ptr->findSecurity(MarketType::Spot, tokens[3]);
112
113 if (security_ptr == nullptr) {
114 throw runtime_error("Security " + tokens[3] + " is not found");
115 }
116
117 // Calling strategy functions
118 vector<std::shared_ptr<Order>> order_vector;
119 MarketType mt = tokens[5] == "S" ? MarketType::Spot : MarketType::Futures;
120 std::shared_ptr<TimeType> tt = std::make_shared<TimeType>(tokens[0]);
121 std::shared_ptr<OrderBook> ob = getOrderbook(mt, *exchange_ptr, *security_ptr);
122
123
124 if (tokens[2] == "T") {
125 last_traded_price[make_pair(mt, security_ptr)] = stod(tokens[6]);
126 vector<tuple<std::shared_ptr<Order>, double, double>> filled_orders = ob->tradeOccurred(last_traded_price[make_pair(mt, security_ptr)], stod(tokens[7]));
127
128 for (auto it : filled_orders) {
129 std::get<0>(it)->fillOrder(std::get<2>(it), std::get<1>(it));
130
131 Trade* trade = new Trade(std::get<0>(it), tt, std::get<0>(it)->getSide(), std::get<2>(it), std::get<1>(it), true);
132 tradelog.addTrade(std::shared_ptr<Trade>(trade));
133
134 user.updateBalance(std::get<0>(it)->getMarketType(), -trade->getSide() * std::get<2>(it) * std::get<1>(it) - trade->getFee());
135 strategy->updatePosition(std::get<0>(it)->getMarketType(), *std::get<0>(it)->getExchange(), *std::get<0>(it)->getSecurity(), std::get<0>(it)->getSide()*std::get<2>(it));
136 }
137
138
139 TradeEventMsg msg(tt, exchange_ptr, mt, security_ptr, ob, stod(tokens[6]), stod(tokens[7]));
140 order_vector = strategy->onTrade(msg);
141 }
142
143 else if (tokens[2] == "BID_UPDATE" || tokens[2] == "ASK_UPDATE") {
144 vector<tuple<std::shared_ptr<Order>, double, double>> filled_orders = tokens[2] == "BID_UPDATE" ? ob->buySideUpdated(stod(tokens[8]), stod(tokens[9]))
145 : ob->sellSideUpdated(stod(tokens[14]), stod(tokens[15]));
146
147 for (auto it : filled_orders) {
148 std::get<0>(it)->fillOrder(std::get<2>(it), std::get<1>(it));
149
150 Trade* trade = new Trade(std::get<0>(it), tt, std::get<0>(it)->getSide(), std::get<2>(it), std::get<1>(it), true);
151 tradelog.addTrade(std::shared_ptr<Trade>(trade));
152
153 user.updateBalance(std::get<0>(it)->getMarketType(), -trade->getSide() * std::get<2>(it) * std::get<1>(it) - trade->getFee());
154 strategy->updatePosition(std::get<0>(it)->getMarketType(), *std::get<0>(it)->getExchange(), *std::get<0>(it)->getSecurity(), std::get<0>(it)->getSide()*std::get<2>(it));
155 }
156
157 QuoteEventMsg msg(tt, exchange_ptr, mt, security_ptr, ob, stod(tokens[8]), stod(tokens[9]), stod(tokens[14]), stod(tokens[15]));
158 order_vector = strategy->onTopQuote(msg);
159 }
160
161 else if (tokens[2] == "BUY_SIDE_UPDATE" || tokens[2] == "SELL_SIDE_UPDATE") {
162 vector<tuple<std::shared_ptr<Order>, double, double>> filled_orders = tokens[2] == "BUY_SIDE_UPDATE" ? ob->buySideUpdated(stod(tokens[6]), stod(tokens[7]))
163 : ob->sellSideUpdated(stod(tokens[6]), stod(tokens[7]));
164
165 for (auto it : filled_orders) {
166 std::get<0>(it)->fillOrder(std::get<2>(it), std::get<1>(it));
167
168 Trade* trade = new Trade(std::get<0>(it), tt, std::get<0>(it)->getSide(), std::get<2>(it), std::get<1>(it), true);
169 tradelog.addTrade(std::shared_ptr<Trade>(trade));
170
171 user.updateBalance(std::get<0>(it)->getMarketType(), -trade->getSide() * std::get<2>(it) * std::get<1>(it) - trade->getFee());
172 strategy->updatePosition(std::get<0>(it)->getMarketType(), *std::get<0>(it)->getExchange(), *std::get<0>(it)->getSecurity(), std::get<0>(it)->getSide()*std::get<2>(it));
173 }
174
175 DepthEventMsg msg(tt, exchange_ptr, mt, security_ptr, ob, tokens[2] == "BUY_SIDE_UPDATE" ? 1 : -1, stod(tokens[6]), stod(tokens[7]));
176 order_vector = strategy->onDepth(msg);
177 }
178
179 // Append order to the back of the current order
180 if (!order_vector.empty()) {
181 for (auto&& it : order_vector) {
182 map<MarketType, double> avail_buying_pwr = {{MarketType::Spot, user.getCapital(MarketType::Spot)},{MarketType::Futures, user.getCapital(MarketType::Futures)}};
183 avail_buying_pwr[it->getMarketType()] -= it->getBaseCurrencySize() * it->getPrice();
184 if (avail_buying_pwr[it->getMarketType()] < 0) {
185 throw std::runtime_error("Submitted order exceeds the available buying power of " + to_string(avail_buying_pwr[it->getMarketType()]));
186 it->rejectOrder();
187 } else {
188 current_orders.push_back(it);
189 orderlog.addOrder(it);
190 }
191 }
192 }
193
194 // Work with orders
195 for (auto&& it : current_orders) {
196 if (it->getOrderState() == OrderState::SentToExchange) {
197 it->checkOrderReceived(*tt);
198 }
199 if (it->isLiveOrder() && (it->getOrderType() == "STOP" || it->getOrderType() == "STOPLIMIT")) {
200 it->checkTriggered(last_traded_price[make_pair(mt, security_ptr)]);
201 }
202 if (it->checkFillability(ob->getBestBid(),ob->getBestAsk())) {
203 pair<std::shared_ptr<Order>, vector<pair<double, double>>> fills;
204
205 if (it->getOrderType() == "MARKET" || it->getOrderType() == "STOP") {
206 fills = ob->fillMarketOrder(it);
207 } else if (it->getOrderType() == "LIMIT" || it->getOrderType() == "STOPLIMIT") {
208 double qty_fillable = min(ob->getLimitInstantFillQuantity(it->getPrice(), it->getSide()), it->getLeverageAdjustedBaseCurrencySize());
209 if (qty_fillable != 0) { fills = ob->instantFillLimit(it, qty_fillable); }
210 if (qty_fillable < it->getLeverageAdjustedBaseCurrencySize()) {
211 ob->addOrder(it->getPrice(), it->getSide(), it->getLeverageAdjustedBaseCurrencySize() - qty_fillable, it);
212 }
213 }
214
215 for (auto& fill_pair : fills.second) {
216 it->fillOrder(fill_pair.second, fill_pair.first);
217
218 if (it->getSide() == 1) {
219 strategy->updatePosition(it->getMarketType(), *exchange_ptr, *security_ptr, fill_pair.second);
220 } else {
221 strategy->updatePosition(it->getMarketType(), *exchange_ptr, *security_ptr, -fill_pair.second);
222 }
223
224 Trade* trade = new Trade(it, tt, it->getSide(), fill_pair.second, fill_pair.first, false);
225 tradelog.addTrade(std::shared_ptr<Trade>(trade));
226
227 user.updateBalance(it->getMarketType(), -trade->getSide() * fill_pair.first * fill_pair.second - trade->getFee());
228 }
229 }
230 }
231
232 // Remove not-live orders
233 for (auto it = current_orders.begin(); it != current_orders.end();) {
234 if (!(*it)->isLiveOrder() && !((*it)->getOrderState() == OrderState::SentToExchange)) {
235 it = current_orders.erase(it);
236 } else {
237 ++it;
238 }
239 }
240
241 // Record balance history
242 if (tradelog.getTrades().empty()) {
243 tradelog.addBalanceHistory(tt, user.getCapital(MarketType::Spot), user.getCapital(MarketType::Futures));
244 } else {
245 double spot_cap = user.getCapital(MarketType::Spot);
246 double futures_cap = user.getCapital(MarketType::Futures);
247
248 for (const auto& pair : strategy->getPositionMap()) {
249 if (pair.second != 0) {
250 if (std::get<0>(pair.first) == MarketType::Spot) {
251 double pos = strategy->getPosition(MarketType::Spot, *exchange_ptr, *security_ptr);
252 spot_cap += pos * (tradelog.computeWeightedAverageFillPrice(abs(pos)) - last_traded_price[{MarketType::Spot, security_ptr}]);
253 } else {
254 double pos = strategy->getPosition(MarketType::Futures, *exchange_ptr, *security_ptr);
255 spot_cap += pos * (tradelog.computeWeightedAverageFillPrice(abs(pos)) - last_traded_price[{MarketType::Futures, security_ptr}]);
256 }
257 }
258 }
259
260 tradelog.addBalanceHistory(tt, spot_cap, futures_cap);
261 }
262 }

270 void run_latency_analysis(const string& data_path, const string& output_path) {
271 map<MarketType, double> initial_buying_power = {{MarketType::Spot, user.getCapital(MarketType::Spot)}, {MarketType::Futures, user.getCapital(MarketType::Futures)}};
272 vector<int> latency_values = {0, 10, 25, 50, 100, 200, 500, 1000};
273
274 for (auto latency : latency_values) {
275 Clear(initial_buying_power); // Reset
276
277 // Set latency value
278 for (auto exchange : user.getExchanges()) {
279 exchange->setSendingLatency(latency);
280 }
281
282 runBacktest(data_path);
283 latency_analysis_pnl.emplace_back(make_pair(latency, make_pair(tradelog.getBalanceHistory().back().second.first, tradelog.getBalanceHistory().back().second.second)));
284 }
285

292 outfile << it.first << "," << it.second.first << "," << it.second.second << "\n";
293 }
