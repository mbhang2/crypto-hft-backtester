106 pair<std::shared_ptr<Order>, vector<pair<double, double>>> instantFillLimit(std::shared_ptr<Order> order_ptr, double qty) {
107 if (!order_ptr->isLiveOrder()) {
108 return pair<std::shared_ptr<Order>, vector<pair<double, double>>>();
109 }
110
111 double quantity_to_fill = qty;
112 int side = order_ptr->getSide();
113
114 double best_price = (side == 1) ? getBestAsk() : getBestBid();
115 double not_our_order_total_size = getLevelNotOurOrderTotalSize(best_price);
116
117 std::vector<std::pair<double, double>> fills;
118
119 if (quantity_to_fill <= not_our_order_total_size) {
120 // Can fill entirely at the best price level
121 reduceOrder(best_price, quantity_to_fill);
122 fills.emplace_back(best_price, quantity_to_fill);
123 } else {
124 // Partial fill at the best price level
125 reduceOrder(best_price, not_our_order_total_size);
126 if (not_our_order_total_size != 0) {
127 fills.emplace_back(best_price, not_our_order_total_size);
128 }
129 quantity_to_fill -= not_our_order_total_size;
130
131 double next_level = side == 1 ? getNextSellSideLevel(best_price) : getNextBuySideLevel(best_price);
132
133 while (next_level != -1 && quantity_to_fill > 0) {
134 not_our_order_total_size = getLevelNotOurOrderTotalSize(next_level);
135 double subtracting = min(quantity_to_fill, not_our_order_total_size);
136 reduceOrder(next_level, subtracting);
137 if (not_our_order_total_size != 0) {
138 fills.emplace_back(next_level, subtracting);
139 }
140 quantity_to_fill -= subtracting;
141
142 next_level = side == 1 ? getNextSellSideLevel(next_level) : getNextBuySideLevel(next_level);
143 }
144
145 fills.back().second += quantity_to_fill;
146 }

209 vector<tuple<std::shared_ptr<Order>, double, double>> filled_orders;
210 addLevel(price_level, 1); // Handling case where price level does not exist
211
212 // First update the price level
213 if (book[price_level].first == 1) {
214 double not_our_order_size = getLevelNotOurOrderTotalSize(price_level);
215 double our_order_size = getLevelOurOrderTotalSize(price_level);
216
217 if (order_size > not_our_order_size + our_order_size) {
218 // Add order if updated order size is greater than original
219 addOrder(price_level, 1, order_size - getLevelTotalSize(price_level), nullptr);
220 } else if (order_size < not_our_order_size + our_order_size) {
221 // Reduce order if updated order size is lesser than original
222 reduceOrder(price_level, min(not_our_order_size, not_our_order_size + our_order_size - order_size));

224 } else {

226 // Fill our orders

493 void reduceOrder(double price_level, double num_to_reduce) {
494 if (book.find(price_level) != book.end()) {
495 auto& order_queue = book.at(price_level).second;
496
497 // Convert the queue to a deque
498 std::deque<std::pair<double, std::shared_ptr<Order>>> order_deque;
499 while (!order_queue.empty()) {
500 order_deque.push_back(order_queue.front());
501 order_queue.pop();
502 }
503
504 // Iterate through the deque from back to front
505 for (auto it = order_deque.rbegin(); it != order_deque.rend() && num_to_reduce > 0; ++it) {
506 auto& order_pair = *it;
507 if (order_pair.second == nullptr) {
508 // Subtract min(double in queue, num_to_reduce)
509 double reduction = std::min(order_pair.first, num_to_reduce);
510 order_pair.first -= reduction;
511 num_to_reduce -= reduction;
512
513 if (order_pair.first == 0) {
514 // Remove entry if quantity becomes 0
515 order_deque.pop_back();
516 }
517 }
518 }
519
520 // Store the updated deque back into the queue
521 for (const auto& order_pair : order_deque) {
522 order_queue.push(order_pair);
523 }
524 }
